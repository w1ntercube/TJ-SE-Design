CREATE TABLE USER (
    userID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(255) NOT NULL,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    creditScore INT DEFAULT 100,
    role VARCHAR(50) DEFAULT 'customer',
    registrationDate DATETIME DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE CATEGORY (
    categoryID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT
);
CREATE TABLE INSTRUMENT (
    instrumentID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    categoryID INT NOT NULL,
    brand VARCHAR(255),
    condition VARCHAR(255),
    price FLOAT NOT NULL,
    stock INT DEFAULT 0,
    availability BOOLEAN DEFAULT TRUE,
    sellerID INT NOT NULL,
    FOREIGN KEY (categoryID) REFERENCES CATEGORY(categoryID),
    FOREIGN KEY (sellerID) REFERENCES USER(userID)
);
CREATE TABLE ORDER (
    orderID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    buyerID INT, 
    renterID INT, 
    sellerID INT NOT NULL, 
    instrumentID INT NOT NULL, 
    orderType VARCHAR(50) NOT NULL, （purchase/rental）
    rentalStartDate DATETIME, 
    rentalEndDate DATETIME,
    deposit FLOAT, 
    orderDate DATETIME DEFAULT CURRENT_TIMESTAMP, 
    status VARCHAR(50) DEFAULT 'pending', 
    totalAmount FLOAT NOT NULL, 
    FOREIGN KEY (buyerID) REFERENCES USER(userID),
    FOREIGN KEY (renterID) REFERENCES USER(userID),
    FOREIGN KEY (sellerID) REFERENCES USER(userID),
    FOREIGN KEY (instrumentID) REFERENCES INSTRUMENT(instrumentID)
);
CREATE TABLE PAYMENT (
    paymentID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    orderID INT NOT NULL, 
    paymentMethod VARCHAR(255) NOT NULL, 
    paymentDate DATETIME DEFAULT CURRENT_TIMESTAMP, 
    paymentStatus VARCHAR(50) DEFAULT 'completed',    paymentAmount FLOAT NOT NULL, 
    FOREIGN KEY (orderID) REFERENCES ORDER(orderID)
);
CREATE TABLE FAVORITE (
    favoriteID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    userID INT NOT NULL,
    instrumentID INT NOT NULL,
    addedDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (userID) REFERENCES USER(userID),
    FOREIGN KEY (instrumentID) REFERENCES INSTRUMENT(instrumentID)
);
CREATE TABLE REVIEW (
    reviewID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    userID INT NOT NULL,
    instrumentID INT NOT NULL,
    rating INT NOT NULL,
    comment TEXT,
    reviewDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'pending',
    FOREIGN KEY (userID) REFERENCES USER(userID),
    FOREIGN KEY (orderID) REFERENCES ORDER(orderID),
    FOREIGN KEY (instrumentID) REFERENCES INSTRUMENT(instrumentID)
);
CREATE TABLE ADMIN_LOG (
    logID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    adminID INT NOT NULL,
    operationType VARCHAR(50) NOT NULL,
    targetType VARCHAR(50),
    targetID INT,
    status VARCHAR(50) DEFAULT 'success',
    description TEXT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP, 
    FOREIGN KEY (adminID) REFERENCES USER(userID)
);
CREATE TABLE TRADE_RULES (
    ruleID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    content TEXT NOT NULL,
    effectiveDate DATETIME NOT NULL,
    lastModifiedDate DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'active'
);
